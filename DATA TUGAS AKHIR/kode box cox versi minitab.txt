# =====================================================================
# 2. FUNGSI LOG-LIKELIHOOD BOX-COX (LOGIKA MINITAB SUBGROUP = 1)
# =====================================================================

def hitung_minitab_llf(lam, y):
    n = len(y)
    # Transformasi Box-Cox Terstandardisasi
    # Menghitung Geometric Mean (Rata-rata Geometrik) dari data asli
    geometric_mean = np.exp(np.mean(np.log(y)))
    
    if lam == 0:
        w = log_y = np.log(y)
        # Jika lambda = 0, bentuk terstandardisasi: w = ln(y) * geometric_mean
        z = log_y * geometric_mean
    else:
        w = (y**lam - 1) / lam
        # Bentuk terstandardisasi untuk lambda != 0
        z = w / (geometric_mean**(lam - 1))
        
    # Hitung Moving Range dari data terstandardisasi (Z)
    moving_range = np.abs(np.diff(z))
    mr_bar = np.mean(moving_range)
    
    # Estimasi Standar Deviasi lokal ala Minitab Control Chart (d2 untuk n=2 adalah 1.128)
    sigma_z = mr_bar / 1.128
    
    # Fungsi Log-Likelihood murni Minitab
    log_likelihood = - (n / 2) * np.log(sigma_z**2)
    return log_likelihood

# =====================================================================
# 3. FUNGSI UNTUK MENCARI LAMBDA OPTIMAL & RENTANG CI 95%
# =====================================================================

def analisis_boxcox_variabel(nama_variabel, data_input):
    # Optimasi nilai lambda (mencari nilai maksimum dengan meminimalkan fungsi negatifnya)
    fungsi_optimasi = lambda l: -hitung_minitab_llf(l[0], data_input)
    hasil_optimasi = minimize(fungsi_optimasi, x0=[1.0], method='Nelder-Mead')
    
    lambda_optimal = hasil_optimasi.x[0]
    max_llf = hitung_minitab_llf(lambda_optimal, data_input)
    
    # Menghitung batas kritis Chi-Square untuk Interval Kepercayaan 95% (df=1)
    # Nilai kritis: Max Log-Likelihood - 1.9207
    nilai_kritis_llf = max_llf - stats.chi2.ppf(0.95, df=1) / 2
    
    # Grid search untuk mencari batas bawah dan batas atas rentang CI
    grup_lambda = np.linspace(-4, 5, 5000)
    llf_grup = [hitung_minitab_llf(l, data_input) for l in grup_lambda]
    
    # Saring nilai lambda yang masuk dalam batas selang kepercayaan
    lambda_lolos_ci = [l for l, llf in zip(grup_lambda, llf_grup) if llf >= nilai_kritis_llf]
    
    batas_bawah = lambda_lolos_ci[0]
    batas_atas = lambda_lolos_ci[-1]
    
    # Print Hasil Analisis
    print(f"=== ANALISIS BOX-COX VARIABEL: {nama_variabel} ===")
    print(f"Optimal Lambda (λ)     : {lambda_optimal:.4f}")
    print(f"Rentang (95% CI)       : [{batas_bawah:.2f} , {batas_atas:.2f}]")
    
    # Cek apakah angka 1 (tanpa transformasi) masuk dalam rentang
    if batas_bawah <= 1.0 <= batas_atas:
        print("Kesimpulan             : Nilai 1 MASUK dalam rentang CI. (Minitab Rounded Value = 1.0 -> Tidak butuh transformasi)")
    else:
        print("Kesimpulan             : Nilai 1 TIDAK MASUK dalam rentang CI. (WAJIB Transformasi/Minitab Rounded Value != 1.0)")
    print("-" * 60 + "\n")
